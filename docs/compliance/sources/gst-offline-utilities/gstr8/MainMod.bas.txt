Public Sub importFromJSON()
    Dim jsonData As Object, filePath As Variant, filePaths() As String, fileProcessed As Integer: fileProcessed = 0
    
        filePaths = FileUtil.openFile(False)
        'Clear all sheets first
        'Call TCSMod.clearTCSSheet
        'Call AmendMod.clearAmendSheet
        'Call HomeMod.clearHomeMod
        
        For Each filePath In filePaths
            If Trim(filePath & vbNullString) <> vbNullString Then
                Set jsonData = CommonUtil.openJSON(CStr(filePath))
                If jsonData.Exists("error_report") Then
                    MsgBox ("Invalid JSON file/format.Please check the file and try opening again."), vbCritical

                    'Call HomeMod.loadGSTINData(jsonData)

'Validates home sheet values
Public Function validateGSTINData() As Boolean
    Sheets("Home").Unprotect "pass"
    isValidated = True
    Dim col As Integer, rowNo As Integer: rowNo = 6
       
    Range(Sheets("Home").Cells(rowNo, 2), Sheets("Home").Cells(rowNo, 5)).Interior.ColorIndex = 2
    Range(Sheets("Home").Cells(rowNo, 2), Sheets("Home").Cells(rowNo, 5)).ClearComments
    
    'GSTIN format check
    
    If Not CommonUtil.matchRegex(Trim(Sheets("Home").Cells(rowNo, 2).value), GSTIN_REGEX_HOME) Then
        Sheets("Home").Cells(rowNo, 2).Interior.ColorIndex = 3
        Sheets("Home").Cells(rowNo, 2).ClearComments
        MsgBox ("Invalid GSTIN in Home page")
        isValidated = False
    End If
    If Trim(Sheets("Home").Cells(rowNo, 4).value) <> vbNullString Then
    ' Clear previous formatting and comments first (if any)
        With Sheets("Home").Cells(rowNo, 4)
            .Interior.ColorIndex = -4142 ' No Fill
            If Not .Comment Is Nothing Then
                .ClearComments
            End If
        End With
        
        ' Ensure the value in "Home" is trimmed of spaces
        Dim cellValue As String
        cellValue = Trim(Sheets("Home").Cells(rowNo, 4).value)
        
        ' Search for the value in "Master sheet"
        With Sheets("Master sheet").Range("$D$3:$D$11")
            Set c = .Find(What:=cellValue, LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=False)
            
            If c Is Nothing Then
                ' If the value is not found, highlight in red and add a comment
                Sheets("Home").Cells(rowNo, 4).Interior.ColorIndex = 3 ' Red color
                Sheets("Home").Cells(rowNo, 4).AddComment "Please select a valid year from the dropdown"
            End If
        End With
    End If

   
     If Trim(Sheets("Home").Cells(rowNo, 5).value) <> vbNullString Then
            With Sheets("Master sheet").Range("$G$2:$G$13")
               Set c = .Find(What:=(Sheets("Home").Cells(rowNo, 5).value))
    '
                    If c Is Nothing Then
                    
                    Sheets("Home").Cells(rowNo, 5).Interior.ColorIndex = 3
                    Sheets("Home").Cells(rowNo, 5).ClearComments
                    Sheets("Home").Cells(rowNo, 5).AddComment "Please select a valid month from the dropdown"
                    
                    
                    
                    End If
            End With
        End If
   
    Sheets("Home").Protect "pass"
        
End Function

Public Function validateHomeGSTINData() As Boolean
    Sheets("Home").Unprotect "pass"
    validateHomeGSTINData = False
    isValidated = True
    Dim col As Integer, rowNo As Integer: rowNo = 6
       
    Range(Sheets("Home").Cells(rowNo, 2), Sheets("Home").Cells(rowNo, 5)).Interior.ColorIndex = xlNone
    Range(Sheets("Home").Cells(rowNo, 2), Sheets("Home").Cells(rowNo, 5)).ClearComments
    
    'GSTIN format check
    
    If Not CommonUtil.matchRegex(Trim(Sheets("Home").Cells(rowNo, 2).value), GSTIN_REGEX_HOME) Then
        Sheets("Home").Cells(rowNo, 2).Interior.ColorIndex = 3
        Sheets("Home").Cells(rowNo, 2).ClearComments
        Sheets("Home").Cells(rowNo, 2).AddComment "Invalid GSTIN"
        validateHomeGSTINData = True
    End If
    'Mandatory field check
    For col = 2 To 5
       If col <> 3 And Trim(Sheets("Home").Cells(rowNo, col).value & vbNullString) = vbNullString Then
         Sheets("Home").Cells(rowNo, col).Interior.ColorIndex = 3
         Sheets("Home").Cells(rowNo, col).ClearComments
        Sheets("Home").Cells(rowNo, col).AddComment "This field is Mandatory"
         validateHomeGSTINData = True
        End If
    Next
    
    
    Sheets("Home").Protect "pass"
End Function
'To check if the tax period in the json is same as the home sheet value
Public Function checkDate(jsonData As Object) As Boolean
 Dim month, year As String
 checkDate = True
 month = Format(Application.VLookup(Trim(Sheets("Home").Cells(6, 5).value), Sheets("Master sheet").Range("G2:H13"), 2, False), "00")
 If (month > 3) Then
    year = Left(Sheets("Home").Cells(6, "D"), 4)
 Else
    year = "20" + Right(Sheets("Home").Cells(6, "D"), 2)
End If
 If (jsonData.Exists("gstin") And Trim(jsonData("gstin") & vbNullString) <> vbNullString) _
        And (jsonData.Exists("fp") And Trim(jsonData("fp") & vbNullString) <> vbNullString) Then
        If Trim(Sheets("Home").Cells(6, 2).value) <> jsonData("gstin") Or (Left(jsonData("fp"), 2)) <> month Or Right(jsonData("fp"), 4) <> year Then
             MsgBox ("The GSTIN, Financial Year or Tax Period entered do not match in opened JSON file.Please check the file and open again.")
             checkDate = False
            End
        
        End If
    Else
        MsgBox "The GSTIN, Financial Year or Tax Period is missing in opened JSON file.Please check the file and open again.", vbCritical
        checkDate = False
        End
    End If
End Function

Public Function createSimpleJson(sheetName As String, startRow As Integer, colStartNum As Integer, _
    colEndNum As Integer, colPrimaryKey As Integer, Optional colUniqueKey1 As Integer, _
    Optional colUniqueKey2 As Integer, Optional colUniqueKey3 As Integer, _
    Optional itemSubName As String) As Scripting.Dictionary
    
    Dim jsonList As New Scripting.Dictionary, jsonObject As New Scripting.Dictionary, jsonItems As New Scripting.Dictionary
    Dim i As Integer, j As Integer, lastRow As Integer
    Dim cell As Variant, mapKeyValue As Variant, doubleVal As Double
    Dim rng As Range
    Dim compositeKey As String
    
    With Sheets("Suppliers Master-RD")
        i = startRow
        Do Until (IsEmpty(.Cells(i, colPrimaryKey)) _
            Or Trim(.Cells(i, colPrimaryKey).value & vbNullString) = vbNullString)
            compositeKey = Trim(.Cells(i, colPrimaryKey).value)
            If colUniqueKey1 <> 0 Then
                compositeKey = compositeKey & "_" & Trim(.Cells(i, colUniqueKey1).value)
            End If
            If colUniqueKey2 <> 0 Then
                compositeKey = compositeKey & "_" & Trim(.Cells(i, colUniqueKey2).value)
            End If
            If colUniqueKey3 <> 0 Then
                compositeKey = compositeKey & "_" & Trim(.Cells(i, colUniqueKey3).value)
            End If
            For j = colStartNum To colEndNum
                On Error Resume Next
                Set rng = Worksheets("master").Range(.Cells(startRow - 1, j).value & "_map")
                If Err.Number = 0 Then
                    If .Cells(startRow - 1, j).value = "pos" And Trim(.Cells(i, j).value & vbNullString) <> vbNullString Then
                        mapKeyValue = Application.VLookup(Left(.Cells(i, j).value, 2) & "C", rng, 1, False)
                        mapKeyValue = Left(mapKeyValue, 2)
                    Else
                        mapKeyValue = Application.VLookup(.Cells(i, j).value, rng, 1, True)
                    End If
                    
                    If IsError(mapKeyValue) Then
                        jsonItems(.Cells(startRow - 1, j).value) = vbNullString
                    Else
                        jsonItems(.Cells(startRow - 1, j).value) = Trim(mapKeyValue)
                    End If
                Else
                    Err.Clear
                    
                        jsonItems(.Cells(startRow - 1, j).value) = Trim(.Cells(i, j).value)
                    
                End If
            Next
            If Len(itemSubName) = 0 Then
                jsonList.Add compositeKey, jsonItems
            Else
                jsonObject.Add itemSubName, jsonItems
                jsonList.Add compositeKey, jsonObject
            End If
   
            Set jsonItems = Nothing
            Set jsonObject = Nothing
            
            i = i + 1
        Loop
    End With
    Set createSimpleJson = jsonList
End Function
