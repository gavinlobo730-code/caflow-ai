'Function that validates TCS Sheet
Public Function validateTCS(source As String) As Boolean
    Application.ScreenUpdating = False
    On Error Resume Next
    Dim col, length1 As Integer
    Dim i, j, gstinCol, posCol, rowNo, lastRow As Long
    Dim igst, cgst, sgst, value As Double
    Dim tflag, gstflag As Boolean
    Dim yr, mnthIndex As Integer
    Dim mnth, gstin, pos, Key As String
    Dim mnthArr As Variant
    Dim ws As Worksheet
    Dim duplicate As Object
    
    Set duplicate = CreateObject("Scripting.Dictionary")
    validateTCS = True
    Sheets("3 TCS").Unprotect "pass"
    rowNo = 6
    'last_row = ExtraMod.getLastRowTCS - 1
    
    Set ws = ThisWorkbook.Sheets("3 TCS")
    gstinCol = 2
    posCol = 4
    lastRow = ws.Cells(ws.Rows.count, gstinCol).End(xlUp).row
    
    'Clears all the previous errors and comments in the sheet
    Sheets("3 TCS").Range("N6:N15006").value = ""
    Sheets("3 TCS").Range("B6:M15006").Interior.Color = xlNone
    Sheets("3 TCS").Range("B6:M15006").ClearComments
    'Checks if datas are proper in home sheet
    If (MainMod.validateHomeGSTINData) Then
      Sheets("Home").Unprotect "pass"
      Sheets("Home").Select
      Sheets("Home").Protect "pass"
      MsgBox "Please enter GSTIN, Tax period and Financial year in Home Page to validate the TCS/TCSA sheet."
      Exit Function
    End If
    Do Until rowNo > lastRow
        tflag = True
        gstflag = True
        'Checks If tax columns are empty,and assigns them as zero
        If IsEmpty(Sheets("3 TCS").Cells(rowNo, 10).value) Then
            Sheets("3 TCS").Cells(rowNo, 10).value = 0
        End If
        If IsEmpty(Sheets("3 TCS").Cells(rowNo, 11).value) Then
            Sheets("3 TCS").Cells(rowNo, 11).value = 0
        End If
        If IsEmpty(Sheets("3 TCS").Cells(rowNo, 12).value) Then
            Sheets("3 TCS").Cells(rowNo, 12).value = 0
        End If
        'Mandatory field check
        col = 2
            If Trim(Sheets("3 TCS").Cells(rowNo, col).value & vbNullString) = vbNullString Then
                
                Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, col).ClearComments
                Sheets("3 TCS").Cells(rowNo, col).AddComment "This field cannot be left blank"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
            End If
        'Mandatory field check for POS dropdown
        yr = Left(Sheets("Home").Range("D6").value, 4)
        If (yr >= 2025) Then
        col = 4
            If Trim(Sheets("3 TCS").Cells(rowNo, col).value & vbNullString) = vbNullString Then
                Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, col).ClearComments
                Sheets("3 TCS").Cells(rowNo, col).AddComment "This field cannot be left blank"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
            End If
        End If
      
        
        If Trim(Sheets("3 TCS").Cells(rowNo, 5).value & vbNullString) = vbNullString And Trim(Sheets("3 TCS").Cells(rowNo, 6).value & vbNullString) = vbNullString And Trim(Sheets("3 TCS").Cells(rowNo, 7).value & vbNullString) = vbNullString And Trim(Sheets("3 TCS").Cells(rowNo, 8).value & vbNullString) = vbNullString Then
                
                Sheets("3 TCS").Cells(rowNo, 5).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 5).ClearComments
                Sheets("3 TCS").Cells(rowNo, 5).AddComment "Any one of the Values is Mandatory"
                Sheets("3 TCS").Cells(rowNo, 6).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 6).ClearComments
                Sheets("3 TCS").Cells(rowNo, 6).AddComment "Any one of the Values is Mandatory"
                Sheets("3 TCS").Cells(rowNo, 7).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 7).ClearComments
                Sheets("3 TCS").Cells(rowNo, 7).AddComment "Any one of the Values is Mandatory"
                Sheets("3 TCS").Cells(rowNo, 8).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 8).ClearComments
                Sheets("3 TCS").Cells(rowNo, 8).AddComment "Any one of the Values is Mandatory"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
                
                
            End If
        
        'Mandatory check for action drop down
        
        If Trim(Sheets("3 TCS").Cells(rowNo, 13).value & vbNullString) = vbNullString Then
                Sheets("3 TCS").Cells(rowNo, 13).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 13).ClearComments
                Sheets("3 TCS").Cells(rowNo, 13).AddComment "Please select an action"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
                
            End If
        
        'GSTIN format check
        If Not CommonUtil.matchRegex(Sheets("3 TCS").Cells(rowNo, 2).value, GSTIN_REGEX) Then
            If Trim(Sheets("3 TCS").Cells(rowNo, 2).value) = vbNullString Then
                Sheets("3 TCS").Cells(rowNo, 2).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 2).ClearComments
                Sheets("3 TCS").Cells(rowNo, 2).AddComment "This field cannot be left blank"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
                gstflag = False
                GoTo woded
            Else
                Sheets("3 TCS").Cells(rowNo, 2).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 2).ClearComments
                Sheets("3 TCS").Cells(rowNo, 2).AddComment "Invalid GSTIN"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
                gstflag = False
                GoTo woded
            End If
            
        End If
        'GSTIN deductor check
         If Not MainMod.DeductorCheck(Sheets("3 TCS").Cells(rowNo, 2).value) Then
            Sheets("3 TCS").Cells(rowNo, 2).Interior.ColorIndex = 3
            Sheets("3 TCS").Cells(rowNo, 2).ClearComments
            Sheets("3 TCS").Cells(rowNo, 2).AddComment "Supplier GSTIN can’t be same as Tax collector GSTIN in Home sheet"
            Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
            validateTCS = False
            gstflag = False
        End If
        If Not MainMod.samestateCheck(Sheets("3 TCS").Cells(rowNo, 2).value) Then
            Sheets("3 TCS").Cells(rowNo, 2).Interior.ColorIndex = 3
            Sheets("3 TCS").Cells(rowNo, 2).ClearComments
            Sheets("3 TCS").Cells(rowNo, 2).AddComment "Supplier State Code should be same as Tax collector State Code in Home sheet"
            Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
            validateTCS = False
            gstflag = False
        End If
        
        
woded:
        
        For col = 5 To 9
            
            
                length1 = 13
            
                
            value1 = Trim(Sheets("3 TCS").Cells(rowNo, col).value)
            If Trim(value1) <> vbNullString Then
            
                On Error GoTo Flow
                If IsNumeric(value1) Then
                    'length check
                    If (Len(Split(value1, ".", 2)(0)) > length1) Then
                        Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                        Sheets("3 TCS").Cells(rowNo, col).ClearComments
                        Sheets("3 TCS").Cells(rowNo, col).AddComment "Amount should not be greater than " & CStr(length1) & " digits"
                        Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                        validateTCS = False
                        tflag = False
                    End If
Flow:
                    'Amount check for original
                    If Not CommonUtil.matchRegex(Trim(value1), AMT_REGEX) And Not (col = 9) Then
                        'Negative check
                        If Trim(Sheets("3 TCS").Cells(rowNo, col).value < 0) Then
                            Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                            Sheets("3 TCS").Cells(rowNo, col).ClearComments
                            Sheets("3 TCS").Cells(rowNo, col).AddComment "Invalid Amount-1"
                            Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                            validateTCS = False
                            tflag = False
                        Else
                            
                            Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                            Sheets("3 TCS").Cells(rowNo, col).ClearComments
                            Sheets("3 TCS").Cells(rowNo, col).AddComment "Amount should not have more than 2 decimal places."
                            Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                            validateTCS = False
                            tflag = False
                        End If
        
                    End If
                    
                    If Not CommonUtil.matchRegex(Trim(value1), AMT_REGEX_1) And (col = 9) Then
                                                  
                            Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                            Sheets("3 TCS").Cells(rowNo, col).ClearComments
                            Sheets("3 TCS").Cells(rowNo, col).AddComment "Invalid Amount. Amount should not have more than 2 decimal places."
                            Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                            validateTCS = False
                            tflag = False
                              
                    End If
                Else
deci:
                    Sheets("3 TCS").Cells(rowNo, col).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, col).ClearComments
                    Sheets("3 TCS").Cells(rowNo, col).AddComment "Invalid Amount"
                    Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                    validateTCS = False
                    tflag = False
                End If
            
        End If
        Next
        If tflag = False Then
            GoTo wotax
        End If

        ' Tax check Code
           If tflag = True Then
            
            'Calculates net amount
            If Trim(Sheets("3 TCS").Cells(rowNo, 5).value) = vbNullString And Trim(Sheets("3 TCS").Cells(rowNo, 6).value) = vbNullString And Trim(Sheets("3 TCS").Cells(rowNo, 7).value) = vbNullString And Trim(Sheets("3 TCS").Cells(rowNo, 8).value) = vbNullString Then
                Sheets("3 TCS").Cells(rowNo, 9).ClearContents
            ElseIf (Trim(Sheets("3 TCS").Cells(rowNo, 5).value) = vbNullString Or IsNumeric(Trim(Sheets("3 TCS").Cells(rowNo, 5).value))) And (Trim(Sheets("3 TCS").Cells(rowNo, 6).value) = vbNullString Or IsNumeric(Trim(Sheets("3 TCS").Cells(rowNo, 6).value))) And (Trim(Sheets("3 TCS").Cells(rowNo, 7).value) = vbNullString Or IsNumeric(Trim(Sheets("3 TCS").Cells(rowNo, 7).value))) And (Trim(Sheets("3 TCS").Cells(rowNo, 8).value) = vbNullString Or IsNumeric(Trim(Sheets("3 TCS").Cells(rowNo, 8).value))) Then
                  Sheets("3 TCS").Cells(rowNo, 9) = Application.Round((Sheets("3 TCS").Cells(rowNo, 5) + Sheets("3 TCS").Cells(rowNo, 7) - Sheets("3 TCS").Cells(rowNo, 6) - Sheets("3 TCS").Cells(rowNo, 8)), 2)
                 
            Else
               Sheets("3 TCS").Cells(rowNo, 9).ClearContents
            End If
            
            homevalue = CStr(Sheets("Home").Cells(6, 2).value)
            sheetvalue = CStr(Sheets("3 TCS").Cells(rowNo, 2).value)
            
            igst = Application.Round(Sheets("3 TCS").Cells(rowNo, 10).value, 2)
            cgst = Application.Round(Sheets("3 TCS").Cells(rowNo, 11).value, 2)
            sgst = Application.Round(Sheets("3 TCS").Cells(rowNo, 12).value, 2)
            
            If Sheets("3 TCS").Cells(rowNo, 9).value > 0 Then
             
             
             If igst + cgst + sgst = 0 Or IsEmpty(igst + cgst + sgst) Then
                Sheets("3 TCS").Cells(rowNo, 10).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 11).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 12).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 10).ClearComments
                Sheets("3 TCS").Cells(rowNo, 11).ClearComments
                Sheets("3 TCS").Cells(rowNo, 12).ClearComments
                Sheets("3 TCS").Cells(rowNo, 10).AddComment "One of either Integrated Tax or CGST/SGST applicable"
                Sheets("3 TCS").Cells(rowNo, 11).AddComment "One of either Integrated Tax or CGST/SGST applicable"
                Sheets("3 TCS").Cells(rowNo, 12).AddComment "One of either Integrated Tax or CGST/SGST applicable"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
             End If
                'rowNo = rowNo + 1
                'GoTo check
            If cgst <> sgst Then
                        Sheets("3 TCS").Cells(rowNo, 11).Interior.ColorIndex = 3
                        Sheets("3 TCS").Cells(rowNo, 12).Interior.ColorIndex = 3
                        Sheets("3 TCS").Cells(rowNo, 11).ClearComments
                        Sheets("3 TCS").Cells(rowNo, 12).ClearComments
                        Sheets("3 TCS").Cells(rowNo, 11).AddComment "CGST and SGST amount should be equal"
                        Sheets("3 TCS").Cells(rowNo, 12).AddComment "CGST and SGST amount should be equal"
                        Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                        validateTCS = False
                        'rowNo = rowNo + 1
                        'GoTo check
            Else
                
                yr = Left(Sheets("Home").Range("D6").value, 4)
                mnthArr = Array("April", "May", "June")
                mnthIndex = IsInArray(Sheets("Home").Range("E6").value, mnthArr)
                
                value = Round((igst + cgst + sgst) - Application.Round((Sheets("3 TCS").Cells(rowNo, 9).value) * (Sheets("Master sheet").Cells(3, 6).value), 2), 2)
                
                If (yr < 2024 Or (yr = 2024 And mnthIndex <> -1)) And (igst + cgst + sgst) <> Application.Round((Sheets("3 TCS").Cells(rowNo, 9).value) * (Sheets("Master sheet").Cells(3, 6).value), 2) And Not (value >= -0.01 And value <= 0.01) Then
                    Sheets("3 TCS").Cells(rowNo, 10).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 11).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 12).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 10).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 11).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 12).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 10).AddComment "Total tax value must be 1% of net amount liable for TCS"
                    Sheets("3 TCS").Cells(rowNo, 11).AddComment "Total tax value must be 1% of net amount liable for TCS"
                    Sheets("3 TCS").Cells(rowNo, 12).AddComment "Total tax value must be 1% of net amount liable for TCS"
                    Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                    validateTCS = False
                    'rowNo = rowNo + 1
                    'GoTo check
                Else
                    yr = Left(Sheets("Home").Range("D6").value, 4)
                    mnthArr = Array("January", "February", "March", "July", "August", "September", "October", "November", "December")
                    mnthIndex = IsInArray(Sheets("Home").Range("E6").value, mnthArr)
                
                    netliable = Sheets("3 TCS").Cells(rowNo, 9).value
                    taxAmount = Application.Round((igst + cgst + sgst), 2)
                    expectedTaxAmount1 = Application.Round(netliable * (Sheets("Master sheet").Cells(3, 6).value), 2)
                    expectedTaxAmount2 = Application.Round(netliable * (Sheets("Master sheet").Cells(6, 6).value), 2)
                        If expectedTaxAmount1 > expectedTaxAmount2 Then
                            temp = expectedTaxAmount1
                            expectedTaxAmount1 = expectedTaxAmount2
                            expectedTaxAmount2 = temp
                        End If
                     If (yr > 2024 Or (yr = 2024 And mnthIndex <> -1)) And Not (value >= -0.01 And value <= 0.01) Then
                        If (taxAmount < expectedTaxAmount1 Or taxAmount > expectedTaxAmount2) Then
                        Sheets("3 TCS").Cells(rowNo, 10).Interior.ColorIndex = 3
                        Sheets("3 TCS").Cells(rowNo, 11).Interior.ColorIndex = 3
                        Sheets("3 TCS").Cells(rowNo, 12).Interior.ColorIndex = 3
                        Sheets("3 TCS").Cells(rowNo, 10).ClearComments
                        Sheets("3 TCS").Cells(rowNo, 11).ClearComments
                        Sheets("3 TCS").Cells(rowNo, 12).ClearComments
                        Sheets("3 TCS").Cells(rowNo, 10).AddComment "Total tax value must be between 0.5% to 1% of net amount liable for TCS"
                        Sheets("3 TCS").Cells(rowNo, 11).AddComment "Total tax value must be between 0.5% to 1% of net amount liable for TCS"
                        Sheets("3 TCS").Cells(rowNo, 12).AddComment "Total tax value must be between 0.5% to 1% of net amount liable for TCS"
                        Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                        validateTCS = False
                    End If
                End If
                
                End If
              
            End If
            Else
                If igst > 0 Or cgst > 0 Or sgst > 0 Then
                     Sheets("3 TCS").Cells(rowNo, 10).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 11).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 12).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 10).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 11).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 12).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 10).AddComment "Tax amounts are not applicable, when Net Amount eligible for TCS is zero or negative"
                    Sheets("3 TCS").Cells(rowNo, 11).AddComment "Tax amounts are not applicable, when Net Amount eligible for TCS is zero or negative"
                    Sheets("3 TCS").Cells(rowNo, 12).AddComment "Tax amounts are not applicable, when Net Amount eligible for TCS is zero or negative"
                    Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                    validateTCS = False
                End If
            End If
        End If
        
        
wotax:
        'Duplicate check
        yr = Left(Sheets("Home").Range("D6").value, 4)
        If (yr < 2025) Then
        If gstflag = True Then
            Key = Sheets("3 TCS").Cells(rowNo, 2).value & "_" & Sheets("3 TCS").Cells(rowNo, 13).value
            If duplicate.Exists(Key) Then
                Sheets("3 TCS").Cells(rowNo, 2).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 2).ClearComments
                Sheets("3 TCS").Cells(rowNo, 2).AddComment "Duplicate GSTIN Found"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
            Else
                duplicate.Add Key, iCntr
            End If
        End If
        End If
        
        'Duplicate check for same GSTIN and POS
         'For rowNo = 6 To lastRow
        yr = Left(Sheets("Home").Range("D6").value, 4)
        If (yr >= 2025) Then
        If gstflag = True Then
          Key = ws.Cells(rowNo, gstinCol).value & "_" & ws.Cells(rowNo, posCol).value
            If duplicate.Exists(Key) Then
                Sheets("3 TCS").Cells(rowNo, 2).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 2).ClearComments
                Sheets("3 TCS").Cells(rowNo, 2).AddComment "POS cannot be repeated under same GSTIN"
                Sheets("3 TCS").Cells(rowNo, 4).Interior.ColorIndex = 3
                Sheets("3 TCS").Cells(rowNo, 4).ClearComments
                Sheets("3 TCS").Cells(rowNo, 4).AddComment "POS cannot be repeated under same GSTIN"
                Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                validateTCS = False
            Else
                duplicate.Add Key, rowNo
            End If
        End If
       End If
    'Next rowNo
         
         
         
         
        'To check the values of dropdown for POS column
        If Trim(Sheets("3 TCS").Cells(rowNo, 4).value) <> vbNullString Then
            With Sheets("Master sheet").Range("$M$3:$M$40")
               Set c = .Find(What:=(Sheets("3 TCS").Cells(rowNo, 4).value), LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=True)
    '
                    If c Is Nothing Then
                    Sheets("3 TCS").Cells(rowNo, 4).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 4).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 4).AddComment "Please select a valid POS from the dropdown"
                    Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                    validateTCS = False
                    
                    End If
            End With
        End If
         
         
        'To check the values of dropdown
        If Trim(Sheets("3 TCS").Cells(rowNo, 13).value) <> vbNullString Then
            With Sheets("Master sheet").Range("$C$3:$C$4")
               Set c = .Find(What:=(Sheets("3 TCS").Cells(rowNo, 13).value), LookIn:=xlValues, LookAt:=xlWhole, MatchCase:=True)
    '
                    If c Is Nothing Then
                    Sheets("3 TCS").Cells(rowNo, 13).Interior.ColorIndex = 3
                    Sheets("3 TCS").Cells(rowNo, 13).ClearComments
                    Sheets("3 TCS").Cells(rowNo, 13).AddComment "Please select a valid action from the dropdown"
                    Sheets("3 TCS").Cells(rowNo, 14).value = "Error in row"
                    validateTCS = False
                    
                    End If
            End With
        End If
        
        rowNo = rowNo + 1

Loop
    
    If validateTCS And Not IsEmpty(Sheets("3 TCS").Cells(6, 2).value) And (source <> "summary") Then
        MsgBox "Sheet successfully validated. No error(s) found"
    End If
    
    Sheets("3 TCS").Protect Password:="pass", DrawingObjects:=False, userinterfaceonly:=True
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    
    Worksheets("3 TCS").Activate
End Function

    Dim value1 As Variant
    Dim yr, mnthIndex As Integer

        End If

            Sheets("4 Amend").Cells(rowNo, 16).value = 0
        End If
        If IsEmpty(Sheets("4 Amend").Cells(rowNo, 17).value) Then
            Sheets("4 Amend").Cells(rowNo, 17).value = 0
        End If
        'Throws error if there are data in POS field prior impl
        yr = Left(Sheets("4 Amend").Cells(rowNo, 2).value, 4)
        If (yr < 2025) Then
         For Each col In Array(6, 9)
            If Trim(Sheets("4 Amend").Cells(rowNo, col).value & vbNullString) <> vbNullString Then

                Sheets("4 Amend").Cells(rowNo, col).ClearComments
                Sheets("4 Amend").Cells(rowNo, col).AddComment "POS is not applicable for return periods prior to April 2025"
                Sheets("4 Amend").Cells(rowNo, 19).value = "Error in row"
                validateAmend = False
            End If
         Next col
        End If
        'Mandatory field check
            yr = Left(Sheets("4 Amend").Cells(rowNo, 2).value, 4)
            If (yr >= 2025) Then
         For Each col In Array(6, 9)
            If Trim(Sheets("4 Amend").Cells(rowNo, col).value & vbNullString) = vbNullString Then
                Sheets("4 Amend").Cells(rowNo, col).Interior.ColorIndex = 3
                Sheets("4 Amend").Cells(rowNo, col).ClearComments
                Sheets("4 Amend").Cells(rowNo, col).AddComment "This field cannot be left blank"
                Sheets("4 Amend").Cells(rowNo, 19).value = "Error in row"
                validateAmend = False
            End If
         Next col
        End If
            

        
        col = 7
            If Trim(Sheets("4 Amend").Cells(rowNo, col).value & vbNullString) = vbNullString Then

Public Function IsInArray(stringToBeFound As String, arr As Variant) As Long
Dim i As Long
  ' default return value if value not found in array
  IsInArray = -1
  For i = LBound(arr) To UBound(arr)
    If StrComp(stringToBeFound, arr(i), vbTextCompare) = 0 Then
      IsInArray = i
      Exit For
    End If
  Next i
End Function
