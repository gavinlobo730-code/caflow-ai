Public Sub clearCMPData()
    Application.ScreenUpdating = False
    Sheets("5. CMP-08").Unprotect "Gstr4@321"
    With Sheets("5. CMP-08")
        .Range(.Cells(START_ROW, 2), .Cells(FIXED_END_ROW_TO_CLEAR, 10)).ClearContents
        .Range(.Cells(START_ROW, 14), .Cells(FIXED_END_ROW_TO_CLEAR, 18)).ClearContents
        .Range(.Cells(START_ROW, 4), .Cells(10, 8)).Interior.ColorIndex = 15
        .Range(.Cells(START_ROW, 4), .Cells(10, 8)).Interior.ColorIndex = 15
        .Range(.Cells(START_ROW, 2), .Cells(FIXED_END_ROW_TO_CLEAR, 10)).ClearComments
        .Range(.Cells(START_ROW, 14), .Cells(FIXED_END_ROW_TO_CLEAR, 18)).ClearComments
    End With
    Application.ScreenUpdating = True
    Sheets("5. CMP-08").Protect "Gstr4@321"
End Sub
