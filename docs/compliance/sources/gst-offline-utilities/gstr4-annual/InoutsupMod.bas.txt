Public Sub loadINOUTSUPData()
    Dim allSheetsValid As Boolean: allSheetsValid = True
    If Not HomeMod.validateGSTINData Then
        MsgBox ("Please enter the details in 'Home' Sheet."), vbCritical
        Exit Sub
     ElseIf MainMod.checkEmpty Then
        allSheetsValid = False
        MsgBox "There are no details provided in table 4B, 4C and 4D. Therefore no liability is to be computed."
        Exit Sub
    Else
     ' MsgBox ("Processing may take some time. Please wait..."), vbMsgBoxSetForeground
    Dim actionCode As Integer: actionCode = MsgBox("Processing may take some time. Please wait...", 1, "Choose options")
    End If
        Dim errorSheets As New VBA.Collection
   
    
    If Not HomeMod.validateGSTINData Then
        allSheetsValid = False
        errorSheets.Add "Home"
    End If
    If Not B2BRCMod.validateB2BRC Then
        allSheetsValid = False
        errorSheets.Add "4B. B2BRC"
    End If
    If Not URPMod.validateURP Then
        allSheetsValid = False
        errorSheets.Add "4C. URP"
    End If
    If Not IMPS.validateIMPS Then
        allSheetsValid = False
        errorSheets.Add "4D.IMPS"
    End If
    
    If allSheetsValid Then
        Call InoutsupMod.computeClearData
        Call InoutsupMod.loadTaxValData
        Call InoutsupMod.loadIGSTData
        Call InoutsupMod.loadCGSTData
        Call InoutsupMod.loadSGSTData
        Call InoutsupMod.loadCESSData
     Else
            MsgBox "There are errors in the sheet(s) : " _
                & vbNewLine & vbNewLine & Join(CommonUtil.collectionToArray(errorSheets), vbNewLine) _
                & vbNewLine & vbNewLine & "There are some validation errors in Table 4B, 4C or 4D. Please correct the same before computing liability.", vbCritica
    End If
End Sub

Public Function validateInoutsup() As Boolean
    Application.ScreenUpdating = False
    Sheets("6. Inward outward supplies").Unprotect "Gstr4@321"
    Dim i, j As Integer
    Dim ctinDetails As New Scripting.Dictionary
    Dim actionDetails As New Scripting.Dictionary
    Dim actionMap As New Scripting.Dictionary
    Dim isValidated As Boolean: isValidated = True
    Dim isSame As Boolean: isSame = True
    Range(Sheets("6. Inward outward supplies").Cells(7, 5), Sheets("6. Inward outward supplies").Cells(11, 5)).Interior.ColorIndex = 2
    Range(Sheets("6. Inward outward supplies").Cells(7, 7), Sheets("6. Inward outward supplies").Cells(11, 8)).Interior.ColorIndex = 2
    Range(Sheets("6. Inward outward supplies").Cells(7, 5), Sheets("6. Inward outward supplies").Cells(11, 10)).ClearComments
    Range(Sheets("6. Inward outward supplies").Cells(7, 10), Sheets("6. Inward outward supplies").Cells(11, 12)).ClearContents
    For i = 7 To 12
                
                 Dim cgstTax As String: cgstTax = Trim(Sheets("6. Inward outward supplies").Cells(i, 7).Value & vbNullString)
                 Dim sgstTax As String: sgstTax = Trim(Sheets("6. Inward outward supplies").Cells(i, 8).Value & vbNullString)
                 If (cgstTax <> sgstTax) Then
                         Sheets("6. Inward outward supplies").Cells(i, 7).Interior.ColorIndex = 3
                         Sheets("6. Inward outward supplies").Cells(i, 7).ClearComments
                         Sheets("6. Inward outward supplies").Cells(i, 7).AddComment "Central and State/UT tax should be equal."
                         Sheets("6. Inward outward supplies").Cells(i, 8).Interior.ColorIndex = 3
                         Sheets("6. Inward outward supplies").Cells(i, 8).ClearComments
                         Sheets("6. Inward outward supplies").Cells(i, 8).AddComment "Central and State/UT tax should be equal."
                         Sheets("6. Inward outward supplies").Cells(i, 10).Value = "Error found in row."
                         isValidated = isValidated And False
                End If
    Next
 
    validateInoutsup = isValidated
    Application.ScreenUpdating = False
    Sheets("6. Inward outward supplies").Protect "Gstr4@321"
End Function

Public Function createOutwardDataJson() As Variant
    Dim items As New Collection
    Dim i As Integer
    For i = 7 To 12
        Application.ScreenUpdating = True
        Sheets("6. Inward outward supplies").Unprotect "Gstr4@321"
        Dim txval As Double, rt As Variant
        Dim d As Object
        Set d = CreateObject("Scripting.Dictionary")
        rt = Sheets("6. Inward outward supplies").Cells(i, 4).Value
        rt = Trim(rt)
        If InStr(rt, "%") > 0 Then
            rt = Replace(rt, "%", "")
            If IsNumeric(rt) Then
                d.Add "rt", CInt(rt)
            Else
                d.Add "rt", 0
            End If
        ElseIf IsNumeric(rt) And rt <> "" Then
            If rt < 1 And rt > 0 Then
                d.Add "rt", CInt(rt * 100)
            Else
                d.Add "rt", CInt(rt)
            End If
        Else
            d.Add "rt", 0
        End If
        txval = Sheets("6. Inward outward supplies").Cells(i, 5).Value
        d.Add "txval", txval
        items.Add d
        Set d = Nothing
        Application.ScreenUpdating = True
        Sheets("6. Inward outward supplies").Protect "Gstr4@321"
    Next
    'MsgBox JsonConverter.ConvertToJson(items, Whitespace:=2)
    Set createOutwardDataJson = items
End Function
