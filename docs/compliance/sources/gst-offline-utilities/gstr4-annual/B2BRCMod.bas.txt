    Set ctinDetails = createSimpleJson("4B. B2BRC", 7, 2, 3, 2, 3, 11)
    Set actionDetails = createSimpleJson("4B. B2BRC", 7, 2, 3, 2, 3)
    
   ' Debug.Print JsonConverter.ConvertToJson(ctinDetails, Whitespace:=2)
    
    
  '  Debug.Print JsonConverter.ConvertToJson(ctinDetails, Whitespace:=2)
    Dim delcount As Integer
    delcount = 15
    For Each k1 In ctinDetails.Keys
        Set itmsCollection = New Collection
         
        For Each k2 In itemsDetails.Keys
'        MsgBox CStr(Split(k2, "_")(0) & Split(k2, "_")(1))
'        MsgBox CStr(k1)
            If CStr(Split(k2, "_")(0) & "_" & Split(k2, "_")(1)) = CStr(Split(k1, "_")(0) & "_" & Split(k1, "_")(1)) Then
                Set itmObject = itemsDetails.Item(k2)
        
                itmsCollection.Add itmObject
                Set itmObject = Nothing
            End If
        Next
        ctinDetails(k1).Add "itms", itmsCollection
      '  Debug.Print JsonConverter.ConvertToJson(ctinDetails, Whitespace:=2)
        Set itmsCollection = Nothing
    Next

    If ctinDetails.count > 0 Then
        Set itmsCollection = New Collection
        For Each k1 In ctinDetails.Keys
             If CStr(Split(k1, "_")(2)) = "Delete" Then
                  ctinDetails(k1).Add "flag", "D"
               Else
                 ' ctinDetails(k1).Add "flag", "A"
              End If
                  '  MsgBox CStr(delcount)
            Set itmObject = ctinDetails.Item(k1)
            itmsCollection.Add itmObject
            Set itmObject = Nothing
        Next
        b2bDetails.Add "b2br", itmsCollection
        Set itmsCollection = Nothing
    End If
    
       'Debug.Print JsonConverter.ConvertToJson(b2bDetails, Whitespace:=2)
    Set createB2BRCJson = b2bDetails
End Function
